cmake_minimum_required(VERSION 3.22.1)
project(luna_pty C)

if(ANDROID)
    add_library(luna_pty SHARED pty.c pty_jni.c)
    target_compile_options(luna_pty PRIVATE -Wall -Wextra -Werror)
else()
    enable_testing()
    add_executable(pty_test pty.c ../../test/cpp/pty_test.c)
    target_compile_definitions(pty_test PRIVATE LUNA_SHELL_PATH="/bin/sh")
    target_compile_options(pty_test PRIVATE -Wall -Wextra -Werror)
    target_include_directories(pty_test PRIVATE .)
    target_link_libraries(pty_test PRIVATE pthread)
    add_test(NAME local_pty COMMAND pty_test)
    set_tests_properties(local_pty PROPERTIES TIMEOUT 15)
endif()
